the previous CSV layout, and two model PDFs.  This implementation intentionally
keeps the non-root numerical conventions of the supplied code while applying
the revised root coefficients and the corrected resistance sequence around
equations (30)-(33) of ``1-3 spenser(1).pdf``.

Rain treatment used for the baseline table:
    F_root = (F_taproot + F_lateral) * k_wet
    M_33   = M_root + M_weight

Equation (29)'s ``f_m`` lookup is a separate Fan (2022) sensitivity scenario,
not an additional baseline multiplier after equation (30)'s ``k_wet``.  The
``f_m`` column is retained to preserve the requested reference CSV schema, but
it is not applied in this baseline calibration.
"""

def solve_quadratic(a: float, b: float, c: float) -> float:
    disc = b * b - 4.0 * a * c
    if disc < 0:
        return 2.0
    sqrt_d = math.sqrt(disc)
    x1 = (-b + sqrt_d) / (2.0 * a)
    x2 = (-b - sqrt_d) / (2.0 * a)
    candidates = [x for x in (x1, x2) if 0.1 < x < 30.0]
    if not candidates:
        return max(x1, x2) if max(x1, x2) > 0 else 2.0
    return min(candidates) if a < 0 else max(candidates)


def calc_crown_param(height: float, tree_para: dict) -> tuple[float, float]:
    dh = tree_para["dh"]
    dc = tree_para["dc"]
    if dh["type"] == "power":
        dbh_cm = dh["a"] * (height ** dh["b"])
    elif dh["type"] == "inverse_log_power":
        dbh_cm = math.exp((math.log(height) - dh["a"]) / dh["b"])
    else:
        dbh_cm = dh["a"] * height + dh["b"]

    if dc["type"] == "linear":
        crown_width = (dbh_cm - dc["b"]) / dc["a"]
    elif dc["type"] == "power":
        crown_width = (dbh_cm / dc["a"]) ** (1.0 / dc["b"])
    elif dc["type"] == "log":
        crown_width = math.exp((dbh_cm - dc["b"]) / dc["a"])
    elif dc["type"] == "log_power":
        crown_width = math.exp(dc["a"] + dc["b"] * math.log(dbh_cm))
    else:
        crown_width = solve_quadratic(dc["a"], dc["b"], dc["c"] - dbh_cm)

    crown_width = max(0.5, min(25.0, crown_width))
    return crown_width, dbh_cm / 100.0


def calc_wind_moment(
    v_mean: float,
    height: float,
    crown_width: float,
    dbh_m: float,
    tree_para: dict,
) -> float:
    area = crown_width * 0.70 * height * tree_para["kcrown"] * 1.05
    crown_depth = crown_width * 0.75
    h_cg = 0.66 * height
    base_sp = 1.0 - math.exp(-2.0 * crown_depth * tree_para["LAI"] / area)
    if v_mean < 15.0:
        eta_vent = 1.0
    elif v_mean < 25.0:
        eta_vent = 0.5 + 0.5 * base_sp
    else:
        eta_vent = base_sp
    force = 0.5 * RHO_AIR * eta_vent * area * (v_mean**2)

    dbh_cm = dbh_m * 100.0
    n_iter = 3 if dbh_cm < 20.0 else (2 if dbh_cm < 35.0 else 1)
    fi, hi = force, h_cg
    # Retained from the supplied implementation so the requested revision is
    # isolated to the stated root/base changes.
    e_dummy = 1.0e9
    for _ in range(n_iter):
        delta_i = fi * (hi**3) / (3.0 * e_dummy)
        alpha_i = math.atan(delta_i / hi)
        fi = fi * math.cos(alpha_i)
        hi = hi + delta_i

    eta_crown = math.exp(-tree_para["beta"] * v_mean) if v_mean >= 25.0 else 1.0
    return fi * eta_crown * hi


def calc_stem_resistance(
    dbh_m: float,
    klive: float,
    kdefect: float,
    omega: float,
    kbase: float,
) -> float:
    dbh_cm = dbh_m * 100.0
    k_size = 0.85 if dbh_cm < 15.0 else (0.90 if dbh_cm < 30.0 else 0.95)
    k_m = math.exp(-2.8 * omega)
    sigma_eff = klive * kdefect * k_size * k_m * SIGMA_B0
    dbh_base_m = 1.05 * dbh_m
    eta_z = ETA_I0 * (1.0 - omega**2)
    z = eta_z * math.pi * (dbh_m**3) / 32.0
    z_base = eta_z * math.pi * (dbh_base_m**3) / 32.0
    m_stem = sigma_eff * z
    # kbase was present in the old factorial table but omitted from the old
    # code path.  It applies only to the base check, matching the supplied
    # model's base-effective-strength definition.
    m_stem_base = sigma_eff * kbase * z_base
    return min(m_stem, m_stem_base)


def calc_root_resistance(
    crown_width: float,
    dbh_m: float,
    site: int,
    rain_config: dict,
    tree_para: dict,
) -> float:
    root = tree_para["root"]
    root_depth = root["root_depth"]
    le = root_depth * root["L_eff_ratio"]
    if site == 0:
        ksyn = KSYN_OPEN
        h_root = 0.5 * root_depth
    else:
        # R_root remains part of the uploaded species parameterization even
        # though the force formula uses the revised equivalent lever arm.
        _limited_root_radius = min(root["R_root"], crown_width * 0.45)
        ksyn = KSYN_PAVED
        h_root = 0.365 * root_depth

    gamma_eff = 17000.0
    mu_eff = 0.32
    taproot_diameter = dbh_m
    f_tap_side = (
        0.5
        * math.pi
        * taproot_diameter
        * gamma_eff
        * mu_eff
        * (le**2)
    )
    f_end = 0.25 * math.pi * (taproot_diameter**2) * gamma_eff * le
    f_tap_total = f_tap_side + f_end

    dbh_cm = dbh_m * 100.0
    lateral_diameter_mm = max(
        LATERAL_ROOT_DIAMETER_MIN_MM,
        min(LATERAL_ROOT_DIAMETER_MAX_MM, dbh_cm * 2.2),
    )
    f_single_lateral = 15.22 * (lateral_diameter_mm**1.61)
    f_lateral = ksyn * KSHEAR * root["n_side"] * f_single_lateral

    # Baseline sequence: kwet once in root force, then the Eq. 33 sum.  The
    # Fan (2022) fm lookup is a separate sensitivity scenario and is not
    # compounded with the baseline kwet reduction.
    f_root = (f_tap_total + f_lateral) * rain_config["kwet"]
    m_root = f_root * h_root

    # The supplied model gives this allometric equation as the preferred
    # fallback when species-specific crown-mass coefficients are unavailable.
    # The old code used rho*ellipsoid-volume while omitting the required crown
    # fill factor lambda, effectively treating the entire crown envelope as
    # solid wood and greatly overstating the stabilizing self-weight moment.
    crown_mass = CROWN_MASS_ALLOMETRY_A * (dbh_cm**CROWN_MASS_ALLOMETRY_B)
    m_weight = crown_mass * GRAVITY * h_root
    m_equation_33 = m_root + m_weight
    return m_equation_33


def solve_critical_v(
    height: float,
    tree_para: dict,
    klive: float,
    kdefect: float,
    omega: float,
    kbase: float,
    site: int,
    rain_config: dict,
) -> tuple[float, float, str, float, float, float, float]:
    crown_width, dbh_m = calc_crown_param(height, tree_para)
    stem_r = calc_stem_resistance(dbh_m, klive, kdefect, omega, kbase)
    root_r = calc_root_resistance(crown_width, dbh_m, site, rain_config, tree_para)
    resistance = min(stem_r, root_r)

    # The supplied script stopped at 40 m/s, which censored very resistant
    # combinations.  Expand the bracket before bisection so reported critical
    # speeds are actual roots rather than the old search ceiling.
    v_low, v_high = 0.0, 40.0
    while (
        calc_wind_moment(v_high, height, crown_width, dbh_m, tree_para) < resistance
        and v_high < 160.0
    ):
        v_high *= 2.0
    if v_high > 160.0:
        v_high = 160.0
    for _ in range(60):
        v_mid = 0.5 * (v_low + v_high)
        wind_moment = calc_wind_moment(v_mid, height, crown_width, dbh_m, tree_para)
        if wind_moment < resistance:
            v_low = v_mid
        else:
            v_high = v_mid
    v_mean = 0.5 * (v_low + v_high)
    v_gust = v_mean * GUST_AMPLIFICATION_FACTOR
    failure_mode = "stem_fracture" if stem_r < root_r else "root_overturn"
    return v_mean, v_gust, failure_mode, stem_r, root_r, crown_width, dbh_m


def generate_rows() -> list[dict]:
    rows: list[dict] = []
    for tree_name, tree_para in TREE_PARAMS.items():
        for height in tree_para["H_list"]:
            for klive in KLIVE_LIST:
                for kdefect in KDEFECT_LIST:
                    for omega in OMEGA_LIST:
                        for kbase in KBASE_LIST:
                            for site in SITE_LIST:
                                for rain in RAIN_CONFIG_LIST:
                                    (
                                        v_mean,
                                        v_gust,
                                        failure_mode,
                                        stem_r,
                                        root_r,
                                        crown_width,
                                        dbh_m,
                                    ) = solve_critical_v(
                                        height,
                                        tree_para,
                                        klive,
                                        kdefect,
                                        omega,
                                        kbase,
                                        site,
                                        rain,
                                    )
                                    rows.append(
                                        {
                                            "tree": tree_name,
                                            "H_m": height,
                                            "klive": klive,
                                            "kdefect": kdefect,
                                            "omega": omega,
                                            "kbase": kbase,
                                            "site": site,
                                            "kwet": rain["kwet"],
                                            "fm": rain["fm"],
                                            "rain_label": rain["label"],
                                            "Cw_m": round(crown_width, 4),
                                            "DBH_m": round(dbh_m, 6),
                                            "vcrit_mean": round(v_mean, 4),
                                            "vcrit_gust": round(v_gust, 4),
                                            "M_stem_resist_Nm": round(stem_r, 2),
                                            "M_root_resist_Nm": round(root_r, 2),
                                            "failure_mode": failure_mode,
                                            "gust_peak_8level": GUST_PEAK_8LEVEL,
                                            "threshold_mean_for_8gust": round(
                                                THRESHOLD_MEAN, 4
                                            ),
                                            "eta_I0": ETA_I0,
                                            "gust_amplify": GUST_AMPLIFICATION_FACTOR,
                                            "risk_status_8gust": (
                                                "fail"
                                                if v_mean <= THRESHOLD_MEAN
                                                else "safe"
                                            ),
                                        }
                                    )
    return rows


def main() -> None:
    root = Path(__file__).resolve().parents[1]
    out_dir = root / "outputs"
    out_dir.mkdir(parents=True, exist_ok=True)
    out_csv = out_dir / "tree_wind_modified_formula33_56160.csv"
    audit_json = out_dir / "tree_wind_modified_formula33_audit.json"

    rows = generate_rows()
    expected = 13 * 4 * 4 * 3 * 5 * 3 * 2 * 3
    if len(rows) != expected:
        raise RuntimeError(f"row-count mismatch: {len(rows)} != {expected}")

    with out_csv.open("w", encoding="utf-8-sig", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=FIELDNAMES)
        writer.writeheader()
        writer.writerows(rows)

    audit = {
        "rows": len(rows),
        "columns": FIELDNAMES,
        "species": list(TREE_PARAMS),
        "expected_rows_formula": "13*4*4*3*5*3*2*3",
        "root_revision": {
            "kshear": KSHEAR,
            "ksyn_open": KSYN_OPEN,
            "ksyn_paved": KSYN_PAVED,
            "lateral_root_diameter_mm": [
                LATERAL_ROOT_DIAMETER_MIN_MM,
                LATERAL_ROOT_DIAMETER_MAX_MM,
            ],
            "end_bearing_included": True,
            "sequence": "Froot=(Ftap+Flateral)*kwet; M33=Mroot+Mweight",
            "kwet_count": 1,
            "fm_count": 0,
            "fm_role": "reference-only column; equation 29 sensitivity is not compounded with baseline equation 30",
        },
        "crown_mass_revision": {
            "formula": "m_kg=0.12*DBH_cm^2.18",
            "reason": "preferred fallback in supplied model; replaces rho*ellipsoid volume with omitted fill factor lambda",
        },
        "stem_revision": "kbase applied once to the base-section capacity check",
        "healthy_definition": {
            "klive": max(KLIVE_LIST),
            "kdefect": max(KDEFECT_LIST),
            "omega": min(OMEGA_LIST),
            "kbase": max(KBASE_LIST),
        },
        "white_ash_assumption": {
            "heights_m": TREE_PARAMS["白蜡"]["H_list"],
            "geometry": "Fraxinus chinensis 95th-quantile ln allometry; user-referenced image unavailable",
            "mechanical_and_root_inputs": "medians/representative values from supplied broad-leaved species",
        },
        "calibration": {
            "target_definition": "root_overturn failures / all 8-level-gust failures",
            "target_share_range": [0.30, 0.50],
            "failed_rows": sum(
                row["risk_status_8gust"] == "fail" for row in rows
            ),
            "root_overturn_failures": sum(
                row["risk_status_8gust"] == "fail"
                and row["failure_mode"] == "root_overturn"
                for row in rows
            ),
        },
        "literature_basis": [
            {
                "source": "Cannon, Barrett & Peterson (2015)",
                "doi": "10.1016/j.foreco.2015.07.014",
                "finding": "uprooting was 45% for Pinus and 50% for other hardwoods in static winching tests",
            },
            {
                "source": "Moore (2000)",
                "doi": "10.1016/S0378-1127(00)00298-X",
                "finding": "uprooting incidence varied from 11% on clay to 92% on non-cohesive soils",
            },
            {
                "source": "Cohen, Schwarz & Or (2011)",
                "doi": "10.1029/2010JF001886",
                "finding": "root diameter distribution and progressive bundle failure materially affect peak pullout force",
            },
        ],
    }
    failed_rows = audit["calibration"]["failed_rows"]
    root_failures = audit["calibration"]["root_overturn_failures"]
    audit["calibration"]["observed_root_share"] = (
        root_failures / failed_rows if failed_rows else None
    )
    audit["calibration"]["within_target"] = (
        failed_rows > 0
        and 0.30 <= audit["calibration"]["observed_root_share"] <= 0.50
    )
    audit_json.write_text(json.dumps(audit, ensure_ascii=False, indent=2), encoding="utf-8")
    print(json.dumps({"csv": str(out_csv), "audit": str(audit_json), "rows": len(rows)}, ensure_ascii=False))


if __name__ == "__main__":
    main()
